### 步骤 4 · 建 api

`backend/app/api/equipment.py`

```python
# encoding: utf-8
"""器材管理接口"""
from fastapi import APIRouter, Depends
from sqlalchemy.orm import Session

from app.core.database import get_db
from app.core.response import ok, fail
from app.core.deps import get_current_user, require_admin
from app.models.user import User
from app.models.equipment import Equipment
from app.schemas.equipment import (
    EquipmentCreate, EquipmentUpdate, EquipmentOut,
)

router = APIRouter(prefix="/equipment", tags=["器材管理"])


@router.get("")
def list_equipment(db: Session = Depends(get_db),
                   _: User = Depends(get_current_user)):
    """所有登录用户都能看"""
    items = db.query(Equipment).order_by(Equipment.id).all()
    return ok([EquipmentOut.model_validate(e).model_dump(mode="json")
               for e in items])


@router.post("")
def create_equipment(body: EquipmentCreate, db: Session = Depends(get_db),
                     _: User = Depends(require_admin)):
    e = Equipment(**body.model_dump())
    db.add(e)
    db.commit()
    db.refresh(e)
    return ok(EquipmentOut.model_validate(e).model_dump(mode="json"), "已添加")


@router.put("/{eid}")
def update_equipment(eid: int, body: EquipmentUpdate,
                     db: Session = Depends(get_db),
                     _: User = Depends(require_admin)):
    e = db.query(Equipment).filter(Equipment.id == eid).first()
    if not e:
        return fail(404, "不存在", 404)
    for k, v in body.model_dump(exclude_unset=True).items():
        setattr(e, k, v)
    db.commit()
    db.refresh(e)
    return ok(EquipmentOut.model_validate(e).model_dump(mode="json"), "已更新")


@router.delete("/{eid}")
def delete_equipment(eid: int, db: Session = Depends(get_db),
                     _: User = Depends(require_admin)):
    e = db.query(Equipment).filter(Equipment.id == eid).first()
    if not e:
        return fail(404, "不存在", 404)
    db.delete(e)
    db.commit()
    return ok(None, "已删除")
```

### 步骤 2 · 建页面

`frontend/src/views/admin/Equipment.vue`

```vue
<template>
  <el-card shadow="never">
    <template #header>
      <div style="display:flex;justify-content:space-between">
        <span>器材管理（{{ list.length }}）</span>
        <el-button type="primary" size="small" @click="openCreate">新增</el-button>
      </div>
    </template>

    <el-table :data="list" v-loading="loading">
      <el-table-column prop="name" label="名称" />
      <el-table-column prop="spec" label="规格" />
      <el-table-column prop="total" label="总数" width="80" />
      <el-table-column prop="available" label="可用" width="80" />
      <el-table-column label="操作" width="160">
        <template #default="{ row }">
          <el-button size="small" link type="primary" @click="openEdit(row)">编辑</el-button>
          <el-button size="small" link type="danger" @click="onDelete(row)">删除</el-button>
        </template>
      </el-table-column>
    </el-table>

    <el-dialog v-model="dialog" :title="editing ? '编辑' : '新增'" width="460px">
      <el-form :model="form" label-width="60px">
        <el-form-item label="名称"><el-input v-model="form.name" /></el-form-item>
        <el-form-item label="规格"><el-input v-model="form.spec" /></el-form-item>
        <el-form-item label="总数"><el-input-number v-model="form.total" :min="0" /></el-form-item>
        <el-form-item label="可用"><el-input-number v-model="form.available" :min="0" /></el-form-item>
      </el-form>
      <template #footer>
        <el-button @click="dialog = false">取消</el-button>
        <el-button type="primary" @click="onSave">保存</el-button>
      </template>
    </el-dialog>
  </el-card>
</template>

<script setup>
import { ref, reactive, onMounted } from 'vue'
import { ElMessage, ElMessageBox } from 'element-plus'
import { listEquipment, createEquipment, updateEquipment, deleteEquipment } from '../../api/equipment'

const list = ref([])
const loading = ref(false)
const dialog = ref(false)
const editing = ref(false)
const editId = ref(null)
const form = reactive({ name: '', spec: '', total: 0, available: 0 })

async function load() {
  loading.value = true
  try { list.value = await listEquipment() } finally { loading.value = false }
}
function openCreate() {
  editing.value = false
  Object.assign(form, { name: '', spec: '', total: 0, available: 0 })
  dialog.value = true
}
function openEdit(row) {
  editing.value = true; editId.value = row.id
  Object.assign(form, row)
  dialog.value = true
}
async function onSave() {
  if (editing.value) await updateEquipment(editId.value, form)
  else await createEquipment(form)
  ElMessage.success('已保存'); dialog.value = false; load()
}
async function onDelete(row) {
  await ElMessageBox.confirm(`删除「${row.name}」？`, '确认', { type: 'warning' })
  await deleteEquipment(row.id); ElMessage.success('已删除'); load()
}
onMounted(load)
</script>
```

## 五、编码规范（保持一致性）

### 后端

```python
# ✅ 好的写法
@router.get("")
def list_xxx(db: Session = Depends(get_db), user: User = Depends(get_current_user)):
    """一句话说明这个接口干什么"""
    items = db.query(Xxx).all()
    return ok([XxxOut.model_validate(i).model_dump(mode="json") for i in items])


# ❌ 差的写法（没注释、没类型、直接 return 对象）
@router.get("")
def list(db):
    return db.query(Xxx).all()
```

**规则：**
```
① 每个接口一行 docstring
② 用 ok() / fail()，不要直接 return dict
③ 用 Pydantic schema 校验，不要手写 if
④ 权限用 Depends(require_admin)，不要手写 if role
```

### 前端

```javascript
// ✅ API 层只做 HTTP
export const listXxx = () => request.get('/xxx')

// ❌ 不要在 API 层做业务判断
export const listXxx = async () => {
  const data = await request.get('/xxx')
  return data.filter(x => x.active)   // ← 这属于业务，应该在页面里
}
```

## 六、要避免的坑

| 坑 | 说明 |
|---|---|
| **忘了在 models/__init__.py import** | Alembic 检测不到新表，迁移生成空脚本 |
| **手写 if role == 'admin'** | 用 `Depends(require_admin)`，统一 |
| **前端直接拼 URL** | 用 `api/xxx.js`，方便以后改前缀 |
| **在 API 层写业务逻辑** | API 只做 HTTP，业务在页面/stores |
| **改了 model 不迁移** | 生产会缺字段，报错 |
| **alembic autogenerate 不检查** | 生成错的脚本，升级后数据乱 |
